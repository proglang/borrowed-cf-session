% Translation into the untyped soup. The bracket notation translates processes,
% B translates binder groups, and L distributes one sequential segment.

\newcommand*\USTransProc[2][\Sub]{\TransP[#1]{#2}}
\newcommand*\USTransBind[4]{\mathcal B(#1;#2,#3,#4)}
\newcommand*\USTransSeq[4]{\mathcal L(#1;#2,#3,#4)}
\newcommand*\USTransExpr[2][\Sub]{\mathcal T(#2;#1)}

\def\RuleCmdPrefix{RuleSoupTrans}
\def\RuleNamePrefix{Tr-}

\DefRule {Exp} {} {
  \USTransProc{\PExp{\E}}
  = \USCfg{\emptyset}{\emptyset}{\USMulti{\USTransExpr{\E}}}
}

\DefRule {Par} {
  \USTransProc{\Proc[1]}=(C_1,S_1,E_1) \\
  \USTransProc{\Proc[2]}=(C_2,S_2,E_2) \\
  C_1\cap C_2=\emptyset
} {
  \USTransProc{\PPar{\Proc[1]}{\Proc[2]}}
  = \USCfg{C_1\cup C_2}{S_1\cup S_2}{E_1\uplus E_2}
}

\DefRule {New} {
  c\text{ fresh} \\
  \USTransBind{\BindGroup[1]}{c^0}{\USUnit}{\USUnit}
    =(\Sub[1],F_0) \\
  \USTransBind{\BindGroup[2]}{c^1}{\USUnit}{\USUnit}
    =(\Sub[2],F_1) \\
  \USTransProc[{\Sub\cup\sigma_1\cup\sigma_2}]{\Proc}
    =(C,S,E)
} {
  \USTransProc{\PNew{\BindGroup[1]}{\BindGroup[2]}{\Proc}}
  = \USCfg{C\cup\{c\}}{\USUpdate{c}{(\mathsf{open},F_0,F_1)}}{E}
}

\DefRule {SeqNil} {} {
  \USTransSeq{\BEmp}{c^s}{u}{v}=\emptyset
}

\DefRule {SeqOne} {} {
  \USTransSeq{\EVar}{c^s}{u}{v}
  =\SubSingle{\EVar}{\USHandle{u}{c^s}{v}}
}

\DefRule {SeqMore} {
  \Bind\ne\BEmp
} {
  \USTransSeq{\BSeq{\EVar}{\Bind}}{c^s}{u}{v}
  =\SubSingle{\EVar}{\USHandle{u}{c^s}{\USUnit}}
   \cup\USTransSeq{\Bind}{c^s}{\USUnit}{v}
}

\DefRule {BindNil} {} {
  \USTransBind{[]}{c^s}{u}{v}=(\emptyset,[])
}

\DefRule {BindOne} {} {
  \USTransBind{[\Bind]}{c^s}{u}{v}
  =(\USTransSeq{\Bind}{c^s}{u}{v},[])
}

\DefRule {BindSep} {
  \BindGroup\ne[] \\
  z\text{ fresh} \\
  \USTransBind{\BindGroup}{c^s}{z}{v}=(\Sub[2],F)
} {
  \USTransBind{\BWithSep{\Bind}{\BindGroup}}{c^s}{u}{v}
  =\bigl(\USTransSeq{\Bind}{c^s}{u}{z}\cup\Sub[2],
    (\USCell{z}{\BindFlag{\Bind}})\cdot F\bigr)
}
