\def\RuleCmdPrefix{RuleSoupTrans}
\def\RuleNamePrefix{Tr-}

\DefRule {Exp} {} {
  \USTransProc{\PExp{\E}}
  = \USCfg{\emptyset}{\emptyset}{\USMulti{\Sub{[\E]}}}
}

\DefRule {Par} {
  \USTransProc{\Proc[1]}=(C_1,S_1,E_1) \\
  \USTransProc{\Proc[2]}=(C_2,S_2,E_2) \\
  C_1\cap C_2=\emptyset
} {
  \USTransProc{\PPar{\Proc[1]}{\Proc[2]}}
  = \USCfg{C_1\cup C_2}{S_1\cup S_2}{E_1\uplus E_2}
}

\DefRule {New} {
  \Fresh{\TEc} \\
  \USTransBind{\BindGroup[1]}{\TEc[0]}{\USUnit}{\USUnit}
    =(\Sub[1],\TFlags[0]) \\
  \USTransBind{\BindGroup[2]}{\TEc[1]}{\USUnit}{\USUnit}
    =(\Sub[2],\TFlags[1]) \\
  \USTransProc[{\Sub\cup\sigma_1\cup\sigma_2}]{\Proc}
    =(C,S,E)
} {
  \USTransProc{\PNew{\BindGroup[1]}{\BindGroup[2]}{\Proc}}
  = \USCfg{C\cup\{\TEc\}}{\USUpdate{\TEc}{(\StateOpen,\TFlags[0],\TFlags[1])}}{E}
}

\DefRule {SeqNil} {} {
  \USTransSeq{\BEmp}{\TEcs}{\U}{\V}=\emptyset
}

\DefRule {SeqOne} {} {
  \USTransSeq{\EVar}{\TEcs}{\U}{\V}
  =\SubSingle{\EVar}{\USHandle[\U]{\TEcs}[\V]}
}

\DefRule*[lab=Tr-SeqMore] {SeqMore} {
  \Bind\ne\BEmp
} {
  \USTransSeq{\BSeq{\EVar}{\Bind}}{\TEcs}{\U}{\V} =
    \SubSingle{\EVar}{\USHandle[\U]{\TEcs}}
      \cup \USTransSeq{\Bind}{\TEcs}{\USUnit}{\V}
}

\DefRule {BindNil} {} {
  \USTransBind{[]}{\TEcs}{\U}{\V}=(\emptyset,[])
}

\DefRule {BindOne} {} {
  \USTransBind{[\Bind]}{\TEcs}{\U}{\V}
  =
  (\USTransSeq{\Bind}{\TEcs}{\U}{\V},[])
}

\DefRule*[lab=Tr-BindSep] {BindSep} {
  \BindGroup\ne[] \\
  \Fresh\TEVarz \\
  \USTransBind{\BindGroup}{\TEcs}{\TEVarz}{\V}=(\Sub[2],\TFlags)
} {
  \USTransBind{\BWithSep{\Bind}{\BindGroup}}{\TEcs}{\U}{\V} = \bigl(
    \USTransSeq{\Bind}{\TEcs}{\U}{\TEVarz}\cup\Sub[2],
    \USCell{\TEVarz}{\BindFlag{\Bind}} \cdot \TFlags
  \bigr)
}
