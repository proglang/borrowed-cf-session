% Algorithmic Expression Typing

\def\RuleNamePrefix{A-}
\def\RuleCmdPrefix{RuleAlg}

% A-Const
\DefRule [\C \notin \{\CSplit,\CSSplit\}] {Const} {
  \CtxEmpty \SubCtx \Ctx \uparrow \CSet[0] \\
  \HasCType \C \T
}{
  \InferType \C \T \EffPure [\CSet[0]]
}

% A-LSplit
\DefRule {LSplit} {
  \CtxEmpty \SubCtx \Ctx \uparrow \CSet[0] \\
  \neg (\S \SEq \SSkip) \\
  \Fresh\TVar
}{
  \InferType {\CSplit_{\S}}
  {\TArr{\MobMobile}{\DirUnord,\EffPure} {\SSeq\S\TVar} {\TPairOrd{\S}{\TVar}}}
    \EffPure
    [\CSet[0]]
}

% A-RSplit
\DefRule {RSplit} {
  \CtxEmpty \SubCtx \Ctx \uparrow \CSet[0] \\
  \Fresh\TVar
}{
  \InferType {\CSSplit_{\S}}
  {\TArr{\MobMobile}{\DirUnord,\EffPure} {\SSeq\S\TVar} {\TPairUnord{\SSeq{\S}{\SDrop}}{\SSeq\SAcq\TVar}}}
    \EffPure
    [\CSet[0]]
}

% A-Var
\DefRule {Var} {
  \CtxVar \EVar \T \SubCtx \Ctx \uparrow \CSet[0]
} {
  \InferType \EVar {\T} \EffPure [\CSet[0]]
}

% A-App
\DefRule {App} {
  \reqDir{\Dir}{\Dir*} \\
  \reqEff{\Dir*}{\Eff[1]}{\Eff[2]} \\
  \CtxAlgJoinCheck{\E[1]}{\E[2]} \\\\
  \InferType [\CtxRestrict{\E[1]}] {\E[1]} {\TArr{\Mob}{\Dir*,\Eff[3]} \T \TU} {\Eff[1]} [\CSet[1]] \\
  \CheckType [\CtxRestrict{\E[2]}] {\E[2]} {\T} {\Eff[2]} [\CSet[2]]
}{
  \InferType
    {\EAppD[\Dir]{\E[1]}{\E[2]}}
    {\TU}
    {\Eff[1] \EffJoin \Eff[2] \EffJoin \Eff[3]}
    [\CSet[0] \CSJoin \CSet[1] \CSJoin \CSet[2]]
}


% A-Annot
\DefRule {Annot} {
  \CheckType \E {\T} {\Eff} [\CSet]
} {
  \InferType {\E : \T} {\T} \Eff [\CSet]
}


% A-Check
\DefRule {Check} {
  \InferType \E {\TU} {\Eff} \\
}{
  \CheckType \E {\T} {\Eff} [\CSet \CSJoin \{\T \CEq \TU\}]
}

% A-Abs
\DefRule {Abs} {
  \CheckType[\CtxVar\EVar\T \CtxJoin \Ctx]
    \E
    {\TU}
    {\Eff*} \\
  \Eff* \le \Eff \\
  \Dir = \DirNone \implies \Unr \Ctx \wedge \Mob = \MobMobile
}{
  \CheckType
    {\EAbs\EVar\E}
    {\TArr{\Mob}{\Dir,\Eff}{\T}{\TU}}
    {\EffPure}
    [\CSet \cup \left\{ \CMobile {\T*} \mid \CtxVar \EVar {\T*} \in \Ctx \wedge \Mob = \MobMobile \right\}]
}

% A-AbsRec
\DefRule {AbsRec} {
  \Eff* \le \Eff \\
  \Unr\Ctx \\\\
  \CheckType[
    \CtxVar f {\TArr{\MobMobile}{\DirNone,\Eff} \T \TU}
      \CtxPar \CtxVar\EVar\T
      \CtxPar \Ctx
  ]
    \E
    {\TU}
    {\Eff*}
}{
  \CheckType
    {\EMu {\EAbs\EVar\E}}
    {\TArr{\MobMobile}{\DirNone,\Eff} \T \TU}
    {\EffPure}
}

% A-Pair
\DefRule {Pair} {
  \reqEff \Dir {\Eff[2]} {\Eff[1]}\\
  \CtxAlgJoinCheck{\E[1]}{\E[2]} \\\\
  \CheckType[\CtxRestrict{\E[1]}]
    {\E[1]}
    {\T}
    {\Eff[1]}
    [\CSet[1]]
  \\
  \CheckType[\CtxRestrict{\E[2]}]
    {\E[2]}
    {\TU}
    {\Eff[2]}
    [\CSet[2]]
} {
  \CheckType
    {\EPair {\E[1]} {\E[2]}}
    {\T \TPair \TU}
    {\Eff[1] \EffJoin \Eff[2]}
    [\CSet[0] \CSJoin \CSet[1] \CSJoin \CSet[2]]
}

% A-Seq
\DefRule {Seq} {
  \CtxAlgSeqCheck{\E[1]}{\E[2]} \\
  \InferType[\CtxRestrict{\E[1]}]
    {\E[1]}
    {\T}
    {\Eff[1]}
    [\CSet[1]]
  \\
  \InferType[\CtxRestrict{\E[2]}]
    {\E[2]}
    {\TU}
    {\Eff[2]}
    [\CSet[2]]
  \\
  \Unr \T
}{
  \InferType
    {\ESeq{\E[1]}{\E[2]}}
    {\TU}
    {\Eff[1] \EffJoin \Eff[2]}
    [\CSet[0] \CSJoin \CSet[1] \CSJoin \CSet[2]]
}

% A-LetPair
\DefRule {LetPair} {
  \CtxAlgPatCheck{\E[1]}{\E[2]} \\
  \InferType[\CtxRestrict{\E[1]}]
    {\E[1]}
    {\T[1] \TPair \T[2]}
    {\Eff[1]}
    [\CSet[1]]
  \\
  \InferType[(\CtxVar x {\T[1]} \CtxJoin \CtxVar y {\T[2]}) \CtxPatOp \CtxRestrict{\E[2]}]
    {\E[2]}
    {\TU}
    {\Eff[2]}
    [\CSet[2]]
}{
  \InferType
     {\ELetPair xy{\E[1]}{\E[2]}}
     {\TU}
     {\Eff[1] \EffJoin \Eff[2]}
     [\CSet[0] \CSJoin \CSet[1] \CSJoin \CSet[2]]
}

% A-Inj
\DefRule {Inj} {
  \CheckType
    {\E}
    {\T[k]}
    {\Eff}
}{
  \CheckType
    {\EInj k \E}
    {\TVariant{\ell:\T[\ell]}}
    {\Eff}
}

% A-Case
\DefRule {Case} {
  \InferType[\CtxRestrict{\E}]
    {\E}
    {\TVariant{\ell:\T[\ell]}}
    {\Eff}
  \\
  \InferType[\CtxVar {\EVar[\ell]} {\T[\ell]} \CtxPatOp \CtxRestrict*{X}]
    {\E[\ell]}
    {\TU[\ell]}
    {\Eff[\ell]}
    [\CSet[\ell]]
  \\
  X = \textstyle\bigcup_{\ell\in L} \Fv(\E[\ell]) \\
  \CtxRestrict{\E} \CtxPatOp \CtxRestrict*{X} \SubCtx \Ctx \uparrow \CSet[0]
  \\
  \ell' \in L
}{
  \InferType
    {\ECase[\ell\in L] \E \ECaseBranchDef}
    {\TU[\ell']}
    {\Eff \EffJoin \Eff[1] \EffJoin \Eff[2]}
    [\CSet[0] \CSJoin \CSet \CSJoin \bigcup_{\ell\in L} \CSet[\ell] \CSJoin \left\{\TU[k] \CEq \TU[k'] \mid k,k' \in L \right\}]
}

\newcommand*\DefinitionsReqDirEff{
  \begingroup
  \newcommand\AND{\mathrm{~and~}}
  \newcommand\OR{\mathrm{~or~}}
  \begin{array}{r@{\;\Longleftrightarrow\;}c}
    \reqDir{\Dir}{\Dir*}
    &
    (\Dir = \DirLeft \AND \Dir* = \DirLeft)
    \OR
    (\Dir = \DirRight \AND \Dir* \ne \DirLeft)
    \\
    \reqEff{\Dir}{\Eff[1]}{\Eff[2]}
    & (\Dir = \DirLeft \Rightarrow \Eff[1] = \EffPure)
    \AND
    (\Dir = \DirRight \Rightarrow \Eff[2] = \EffPure)
  \end{array}
  \endgroup
}
