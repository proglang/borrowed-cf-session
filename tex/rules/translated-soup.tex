\def\RuleCmdPrefix{RuleSoupRed}
\def\RuleNamePrefix{RU-}

\DefRule {Exp} {
  \E \EReduce \E*
} {
  \USRed
    {\USCfg{C}{S}{\USBag{\E}}}
    {\USCfg{C}{S}{\USBag{\E*}}}
}

\DefRule {Fork} {} {
  \USRed
    {\USCfg{C}{S}{\USBag{\ECapp{\USApp{\CFork}{\V}}}}}
    {\USCfg{C}{S}{\USBag{\ECapp{\USUnit}, \USApp{\V}{\USUnit}}}}
}

\DefRule {Discard} {} {
  \USRed
    {\USCfg{C}{S}{\USBag{\ECapp{\USApp{\CDiscard}{\V}}}}}
    {\USCfg{C}{S}{\USBag{\ECapp{\USUnit}}}}
}

\DefRule {New} {
  \Fresh{\TEc,\TEVarz[0],\TEVarz[1]} \\
  h=\USPair
    {\USHandle[\TEVarz[0]]{\TEc[0]}}
    {\USHandle[\TEVarz[1]]{\TEc[1]}} \\\\
  S'=\USUpdate{\TEc}{(
    \StateOpen,
    [\USCell*{\TEVarz[0]}{\FlagDropped}],
    [\USCell*{\TEVarz[1]}{\FlagDropped}]
  )}
} {
  \USRed
    {\USCfg{C}{S}{\USBag{\ECapp{\USApp{\CNew}{\USUnit}}}}}
    {\USCfg{C\cup\{c\}}{S'}{\USBag{\ECapp{h}}}}
}

\DefRule {LSplit} {
  \USOpen
} {
  \USRed
    {\USCfg{C}{S}{\USBag{\ECapp{\EApp{\CSplit}{\USHandle[u]{\TEcs}[\V]}}}}}
    {\USCfg{C}{S}{\USBag{\ECapp{\USPair {\USHandle[u]{\TEcs}} {\USHandle{\TEcs}[\V]}}}}}
}

% Unary remote split allocates its synchronization cell at the front.
\DefRule {RSplit} {
  \Fresh\TEVarz \\
  S'=\USSideUpdate{\USCell{\TEVarz}{\FlagInit} \cdot F} \\\\
  \USOpen \\
  \USFlags=F
} {
  \USRed
    {\USCfg{C}{S}{\USBag{\ECapp{\EApp{\CSSplit}{
      \USHandle[u]{\TEcs}[\V]}}}}}
    {\USCfg{C}{S'}{\USBag{\ECapp{\USPair
      {\USHandle[u]{\TEcs}[\TEVarz]}
      {\USHandle[\TEVarz]{\TEcs}[\V]}}}}}
}

\DefRule {Drop} {
  \USOpen \\
  S'=\USSideUpdate{A \cdot \USCell{\TEVarz}{\FlagDropped} \cdot B} \\\\
  \USFlags=A\cdot \USCell{\TEVarz}{\FlagInit} \cdot B
} {
  \USRed
    {\USCfg{C}{S}{\USBag{\ECapp{\EApp{\CDrop}{
      \USHandle{\TEcs}[\TEVarz]}}}}}
    {\USCfg{C}{S'}{\USBag{\ECapp{\USUnit}}}}
}

\DefRule {Acquire} {
  \USOpen \\
  S'=\USSideUpdate{A\cdot B} \\\\
  \USFlags=A \cdot \USCell{\TEVarz}{\FlagDropped} \cdot B
} {
  \USRed
    {\USCfg{C}{S}{\USBag{\ECapp{\EApp{\CAcq}{
      \USHandle[\TEVarz]{\TEcs}[\V]}}}}}
    {\USCfg{C}{S'}{\USBag{\ECapp{\USHandle*{\TEcs}[\V]}}}}
}

\DefRule {Close} {
  \USOpen\\
  \USFlags[0]=\USFlags[1]=[]
} {
  \USRed
    {\USCfg{C}{S}{\USBag{
      \ECapp[1]{\EApp{\CTerm}{\USHandle{\TEcs}}},
      \ECapp[2]{\EApp{\CWait}{\USHandle{\TEcs*}}}
    }}}
    {\USCfg{C}{\USUpdate{\TEc}{(\StateClosed,[],[])}}{\USBag{
      \ECapp[1]{\USUnit},
      \ECapp[2]{\USUnit}
    }}}
}

\DefRule {Com} {
  \USOpen
} {
  \USRedOne
    {\USCfg{C}{S}{\USBag{
      \ECapp[1]{\EApp{\CSend}{(\USPair{\V}{\USHandle{\TEcs}[u]})}},
      \ECapp[2]{\EApp{\CRecv}{\USHandle{\TEcs*}[w]}}
    }}}
    {\USCfg{C}{S}{\USBag{\ECapp[1]{\USUnit}, \ECapp[2]{\V}}}}
}

\DefRule {Choice} {
  \USOpen
} {
  \USRed
  {
    \USCfg{C}{S}{\USBag{
      \ECapp[1]{\EApp{\CSelect k}{\USHandle{\TEcs}[u]}},
      \ECapp[2]{\EApp{\CBranch}{\USHandle{\TEcs*}[w]}}
    }}
  }
  {
    \USCfg{C}{S}{\USBag{
      \ECapp[1]{\USHandle*{\TEcs}[u]},
      \ECapp[2]{\targettermcolor{\EInj k {\USHandle{\TEcs*}[w]}}}
    }}
  }
}
